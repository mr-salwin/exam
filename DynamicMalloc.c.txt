DynamicMalloc.c
#include<stdio.h>
#include<stdlib.h>
void main(){
int *p;
int s;
printf("Enter the size of the element: ");
scanf("%d",&s);
p=(int*)malloc(s *sizeof(int));
if(p==NULL){
printf("Memory not allocated");
}else{
printf("Enter %d elements: ",s);
for(int i=0;i<s;i++){
scanf("%d",&p[i]);}
printf("The elements of the array are: ");
for(int j=0;j<s;j++){
printf("%d ",p[j]);}
Free(p);}}
