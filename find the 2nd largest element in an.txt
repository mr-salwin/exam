printf("Memory not allocated");
}else{
printf("Enter %d elements: ",s);
for(int i=0;i<s;i++){
scanf("%d",&p[i]);}

for(int i=0;i<s;i++){

if(p[j]<p[j+1]){
int t=p[j];

}}}
